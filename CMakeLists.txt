cmake_minimum_required(VERSION 3.24)
project(persistent_kv LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "This store requires Linux POSIX file APIs. Use Linux or WSL.")
endif()
include(CTest)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(kv src/kv.cpp)
target_include_directories(kv PUBLIC include)
add_executable(kv_cli cli.cpp)
target_link_libraries(kv_cli PRIVATE kv)
if(BUILD_TESTING)
    include(FetchContent)
    FetchContent_Declare(googletest URL https://codeload.github.com/google/googletest/tar.gz/refs/tags/v1.16.0)
    FetchContent_MakeAvailable(googletest)
    add_executable(kv_tests tests.cpp)
    target_link_libraries(kv_tests PRIVATE kv GTest::gtest_main)
    add_test(NAME kv_tests COMMAND kv_tests)
endif()
